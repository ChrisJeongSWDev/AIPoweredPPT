4. Code Implementation Example

Multi-language reference code templates for initializing and controlling the hardware device connection loop.

#include "VEOFocusSDK.h"

int main() {
    VEO_STATUS status = VEO_Focus_Initialize();
    if (status == VEO_SUCCESS) {
        // Move focus position to target grid
        VEO_Focus_SetPosition(5000);
    }
    return 0;
}
