# AI_Powered_PPT — pinned runtime dependencies
#
#   pip install -r requirements.txt
#
# Deliberately small. python-pptx is the only dependency the analyzer (Deliverable C)
# actually needs; everything else belongs to a later phase and is pinned here so the
# environment is reproducible from day one.

# --- core: needed by analyze_pptx_master.py (Phase 0, today) ---------------------------
python-pptx==1.0.2          # PPTX read/write. Brings in lxml + Pillow + XlsxWriter.
lxml==5.3.0                 # Explicit: we parse the theme part's XML by hand, because
                            # python-pptx exposes no theme API at all.
Pillow==11.0.0              # Explicit: required for aspect-fit. Without it,
                            # `picture.image.size` is unavailable and add_picture()
                            # silently STRETCHES images instead of fitting them.

# --- Phase 1: HTML adapter -------------------------------------------------------------
beautifulsoup4==4.12.3      # The corpus nests (<img> inside .photo-desc inside
                            # .photo-placeholder). Regex is not an option.
jsonschema==4.23.0          # Validate IR against ir/schema.json at every boundary.

# --- Phase 1B: the editor ---------------------------------------------------------------

#

# ~150-line pointer-capture engine (static/interactions.js), which is the alternative plan Sec 5.1
