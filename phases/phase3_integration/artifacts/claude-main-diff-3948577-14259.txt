+def _slide_ir_schema() -> dict[str, Any]:
+    """The single-slide subset of ``ir/schema.json``'s ``$defs.slide``, for validating
+    one ``fill_slide`` call's output instead of a whole deck.
+
+    Keeps ``$defs`` alongside the lifted ``slide`` definition so its internal
+    ``#/$defs/...`` refs (``heading``, ``rich_text``, etc.) still resolve -- those are
+    JSON Pointer fragments resolved against whatever document is handed to the validator,

+    ``$defs`` travels with it.
+    """
+    full = load_schema()

+def _format_layout_budget(layout: dict[str, Any]) -> str:
+    """Render one layout's placeholder budgets in the same style as the
+    ``### Content budgets`` section of ``reference/layout_catalog.md``, so the model sees
+    an already-familiar shape instead of a second, differently-worded budget notation.
+    """
+    placeholders = layout.get("placeholders") or []
+    if not placeholders:
+        return "This layout has no text placeholders (decorative/blank layout)."
+
+    lines = []
+    for ph in placeholders:
+        budget = ph.get("content_budget") or {}
+        if not budget.get("computable"):
+            lines.append(f"- `idx {ph.get('idx')}` {ph.get('type')} -- budget not "
+                         "computable for this placeholder; keep content minimal.")
+            continue
+        inches = ph.get("inches") or {}
+        lines.append(
+            f"- `idx {ph.get('idx')}` {ph.get('type')} -- "
+            f"{inches.get('width', 0):.4f}x{inches.get('height', 0):.4f}in @ "
+            f"{budget.get('font_pt', 0):.1f}pt -- "
+            f"max **{budget.get('max_bullets', 1)} bullets** x "
+            f"**{budget.get('max_words_per_bullet', 0)} words** "
+            f"({budget.get('max_chars_per_bullet_latin', 0)} latin / "
+            f"{budget.get('max_chars_per_bullet_cjk', 0)} korean chars per bullet)"
+        )

+    """Everything ``_slide_ir_schema()`` cannot express: cross-field rules (ragged table
