+def _block_plain_text(block: Any) -> str:
+    """The rendered-character text a block owes ``check_budget`` -- the ``str``
+    counterpart of each block's own ``text_length()`` in ``ir/models.py`` (which
+    returns only a count; ``check_budget`` needs the actual characters to tell CJK
+    from Latin)."""
+    if isinstance(block, Heading):
+        return block.text
+    if isinstance(block, Paragraph):
+        return block.text.plain
+    if isinstance(block, Bullets):
+        return "\n".join(item.text.plain for item in block.items)
+    if isinstance(block, Callout):
+        return f"{block.title or ''} {block.text.plain}".strip()
+    if isinstance(block, Table):
+        cells = " ".join(cell.plain for row in block.rows for cell in row)
+        return f"{' '.join(block.columns)} {cells}".strip()
+    if isinstance(block, Code):
+        return max((variant.code for variant in block.variants), key=len, default="")
+    if isinstance(block, Image):
