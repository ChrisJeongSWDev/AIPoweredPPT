@app.post("/api/generate")
async def api_generate(request: Request) -> Response:
    """{"prompt": "..."} in, canvas-ir/1 out. MOCK without a key, LIVE Gemini with one (Phase 4).

    Either way the editor hands what comes back straight to loadIR(), which judges every element
    before it touches the stage -- so a model draft is no more trusted than a hand-edited file. The
    request shape, the loading state and the hand-off are identical in both modes; only the author
    behind the seam changes.
    """
    body, err = await _read_json(request)
    if err is not None:
        return err

    prompt = body.get("prompt") if isinstance(body, dict) else None
    if not isinstance(prompt, str) or not prompt.strip():
        # Refused BEFORE any model call or sleep. No author would have been invoked for an empty body.
        return JSONResponse(
            {"error": "validation", "detail": 'the body must be {"prompt": "<non-empty text>"}'},
            status_code=422,
        )

    pin = request.app.state.pin

    # --- MOCK: no key configured. The prompt is NOT read; a fixed deck comes back 1.5 s late. ------
    if gemini_api_key() is None:
        log.info("mock generate: %d-char prompt (not read -- no model is called)", len(prompt))
        ir = mock_ir(pin)
        try:
            validate(ir)
        except CanvasValidationError as exc:
            log.exception("the MOCK payload does not validate -- a server bug, not a user error")
            return JSONResponse({"error": "mock", "detail": str(exc)}, status_code=500)
        await asyncio.sleep(MOCK_LATENCY_S)
        return JSONResponse(ir)

    # --- LIVE: a real Gemini call. The prompt IS read now. [REDACTED]
    log.info("live generate: %d-char prompt -> %s", len(prompt), gemini_model())
    try:
        ir = await generate_live(prompt.strip(), pin)
    except GeminiError as exc:
        log.warning("live generate failed: %s", exc)
        return JSONResponse({"error": "generate", "detail": str(exc)}, status_code=502)

    # The assembled deck goes through the SAME validator the export path runs. A model can put a box
    # on the logo as easily as a human can; refusing it here means the editor is never handed a draft
    # it could only bounce, and the detail is the same numbered list validate() gives everywhere.
    try:

    except CanvasValidationError as exc:

# --- the live author (Phase 4) [REDACTED]
#
# The environment picks the mode. Everything here is ordinary, testable code EXCEPT _call_gemini(),
# the single seam that touches the network -- the mocking test replaces exactly that function and
# exercises all the assembly (prompt injection, pin injection, validation) around it for real.

def gemini_api_key() -> str | None:
    """The configured key, or None. Whitespace-only is treated as absent -- an empty `export
    GEMINI_API_KEY=` in a shell profile must not flip the server into a mode it cannot serve."""
    key = os.environ.get("GEMINI_API_KEY", "").strip()
    return key or None


def gemini_model() -> str:
    return os.environ.get("GEMINI_MODEL", "").strip() or DEFAULT_GEMINI_MODEL


def server_mode() -> str:
    """'live' when a key is configured, else 'mock'. The editor reads this from /api/context and
    flips its badge; it never infers the mode from behaviour."""
    return "live" if gemini_api_key() else "mock"


def system_instruction() -> str:
    try:
        return PROMPT_PATH.read_text(encoding="utf-8")
    except OSError as exc:
        raise GeminiError(f"the system prompt could not be read at {PROMPT_PATH}: {exc}") from exc


_FENCE_OPEN = re.compile(r"^```(?:json)?\s*", re.IGNORECASE)
_FENCE_CLOSE = re.compile(r"\s*```$")


def _strip_code_fence(text: str) -> str:
    """Tolerate a ```json ... ``` wrapper. response_mime_type='application/json' should stop the
    model producing one, but a stray fence is a formatting slip, not a reason to fail a good deck."""
    t = text.strip()
    if t.startswith("```"):
        t = _FENCE_CLOSE.sub("", _FENCE_OPEN.sub("", t)).strip()
    return t


def _finalize_live_ir(data: Any, pin: dict[str, Any]) -> dict[str, Any]:
    """Overwrite schema, canvas and target from the LIVE PIN -- the model is never trusted with them.

    The Web prompt tells the model to omit target entirely (it has no business choosing our template
    SHA) and to fix schema and canvas to constants. We ENFORCE all three here rather than hope: a
    model that ignores the instruction does not get to author the pin. Everything else -- title and
    elements -- is the model's, and validate() judges it next.
    """
    if not isinstance(data, dict):
        raise GeminiError(f"the model returned a {type(data).__name__}, not a JSON object")
    ir = dict(data)
    ir["schema"] = "canvas-ir/1"
    ir["canvas"] = dict(CANVAS)
    ir["target"] = {
        "template": pin["renderer_template"]["filename"],
        "template_sha256": pin["renderer_template"]["sha256"],
        "layout_index": pin["layout_signature"]["layout_index"],
    }
    ir.setdefault("title", "")
    if not isinstance(ir.get("elements"), list):
        ir["elements"] = []
    return ir


async def _call_gemini(prompt: str) -> str:
    """THE ONE FUNCTION THAT TOUCHES THE NETWORK. The mocking test monkeypatches exactly this.

    Kept thin on purpose. google-genai is imported LAZILY so the renderer, the CLI and mock mode all
    install and run without it (it is only in the optional 'ai' extra).
    """
    try:
        from google import genai
        from google.genai import types
    except ImportError as exc:
        raise GeminiError(
            "GEMINI_API_KEY is set but the google-genai package is not installed. "
            "Install it with:  pip install google-genai"
        ) from exc

    client = genai.Client(api_key=gemini_api_key())
    try:
        resp = await client.aio.models.generate_content(
            model=gemini_model(),
            contents=prompt,
            config=types.GenerateContentConfig(
                system_instruction=system_instruction(),
                response_mime_type="application/json",
            ),
        )
    except Exception as exc:  # the SDK raises a family of its own errors; the editor wants one banner
        raise GeminiError(f"the model call failed: {exc}") from exc

    text = getattr(resp, "text", None)
    if not text or not text.strip():
        raise GeminiError("the model returned an empty response")
    return text


async def generate_live(prompt: str, pin: dict[str, Any]) -> dict[str, Any]:
    """prompt -> canvas-ir/1 for the live author. Raises GeminiError on any failure."""
    raw = await _call_gemini(prompt)
    try:
        data = json.loads(_strip_code_fence(raw))
    except (json.JSONDecodeError, ValueError) as exc:
        raise GeminiError(f"the model did not return valid JSON: {exc}") from exc
    return _finalize_live_ir(data, pin)
```

  /* Strip a ```json ... ``` fence a Web-Gem paste often carries. The backend tolerates the same
   * wrapper; keeping both lenient means a copy straight out of the chat UI Just Works. */
  function stripFence(text) {
    var t = String(text).trim();
    if (t.indexOf('```') === 0) {
      t = t.replace(/^```(?:json)?\s*/i, '').replace(/\s*```$/, '').trim();
    }
    return t;
  }

  /* WEB GEM WORKFLOW (Phase 4). Read the clipboard, parse it, hand it to loadIR() ??the ONE door
   * into the stage, the same judge a file import and the live author go through. No validation of
   * its own. The snapshot/pushUndo pair is the file-import path's, for the same reason: this
   * replaces the whole deck (one undo step) and only the caller knows loadIR() accepted it. */
  async function pasteGemJSON() {
    clearWarnings();
    if (!navigator.clipboard || !navigator.clipboard.readText) {

        'This browser will not hand a script the clipboard. Use ?쏬oad JSON??to open a file instead.');

    }
    var text;
    try {
      text = await navigator.clipboard.readText();
    } catch (e) {
      warn('Could not read the clipboard',

    }
    if (!text || !text.trim()) {
      warn('The clipboard is empty', 'Copy the JSON from your chat first, then press Paste Gem JSON.');

## Assumptions I made (brief was silent on these)
1. **`target` injection.** The Web prompt tells the model to omit `target`, but `schema.json` requires it and `render` needs the SHA for the stale/pin check. So the live path (like `mock_ir`) overwrites `schema`/`canvas`/`target` from the live pin ??the model is never trusted with the template SHA.
2. **Error status.** Live failures return **502** (upstream/model), distinct from the user's own invalid deck (**422**). The editor already renders `data.detail` for any non-OK status.

@app.get("/api/context")
async def api_context(request: Request) -> dict[str, Any]:
    """Everything the editor must NOT hard-code.

    Chiefly the template SHA. If the editor carried a literal SHA, re-pinning the template -- a
    deliberate, reviewed act (Sec 4.3) -- would turn into a silent 409 for every user of every
    stale browser tab, and the fix would be "hard-refresh", which nobody would guess. Serving it
    keeps reference/template_pin.json the single source of truth on both sides of the wire.
    """
    pin = request.app.state.pin
    return {
        "template_sha256": pin["renderer_template"]["sha256"],
        "layout_index": pin["layout_signature"]["layout_index"],
        "canvas": dict(CANVAS),
        "brand": {"red": f"#{BRAND_RED}", "forbidden": f"#{FORBIDDEN}"},
        "fonts": {"latin": LATIN, "ea": EA},
        "mode": server_mode(),
        "model": gemini_model() if server_mode() == "live" else None,
    }


@app.post("/api/export")
async def api_export(request: Request) -> Response:
    """canvas-ir/1 in, .pptx bytes out. In memory, start to finish.

    The error mapping is the whole job, and the ORDER of the except clauses is the load-bearing
    part: StalePayloadError SUBCLASSES CanvasValidationError, so catching the parent first would
    report every stale payload as a 422 and tell the user to fix a deck that is perfectly fine. It
    is caught first, and it stays first.
    """
    ir, err = await _read_json(request)
    if err is not None:
        return err

    try:
        prs = render(ir)  # validates, checks the pin, checks the SHA, draws
    except StalePayloadError as exc:  # MUST precede CanvasValidationError -- it is a subclass
        return JSONResponse({"error": "stale", "detail": str(exc)}, status_code=409)
    except CanvasValidationError as exc:
        return JSONResponse({"error": "validation", "detail": str(exc)}, status_code=422)
    except TemplatePinError as exc:
        # Unreachable if the startup gate ran -- the template cannot drift mid-process without
        # someone editing it underneath us. Log it loudly rather than pretend it cannot happen.
        log.exception("template pin failed at REQUEST time; the startup gate should have caught it")
        return JSONResponse({"error": "template", "detail": str(exc)}, status_code=500)

    buf = io.BytesIO()
    prs.save(buf)  # in memory. No temp file, no output directory, nothing on disk.

    return Response(
        content=buf.getvalue(),
        media_type=PPTX_MIME,
        headers={"Content-Disposition": content_disposition(ir["title"])},
    )
