+def _decode_images(
+    images_raw: Any,
+) -> tuple[list[dict[str, Any]] | None, JSONResponse | None]:
+    """Validate and base64-decode /api/generate's optional "images" field. Returns (images, None) or
+    (None, error). Called ONLY from the LIVE branch -- see api_generate's docstring for why a
+    malformed value is never this function's problem in mock mode.
+    """
+    if images_raw is None:
+        return None, None
+    if not isinstance(images_raw, list):
+        return None, JSONResponse(
+            {"error": "validation", "detail": '"images" must be a list of {"mime", "data"} objects'},
+            status_code=422,
+        )
+
+    decoded: list[dict[str, Any]] = []
+    for i, img in enumerate(images_raw):
+        if (
+            not isinstance(img, dict)
+            or not isinstance(img.get("mime"), str)
+            or not isinstance(img.get("data"), str)
+        ):
+            return None, JSONResponse(
+                {
+                    "error": "validation",
+                    "detail": f'images[{i}] must be {{"mime": "<type>", "data": "<base64>"}}',
+                },
+                status_code=422,
+            )
+        try:
+            data = base64.b64decode(img["data"], validate=True)
+        except binascii.Error as exc:
+            return None, JSONResponse(
+                {"error": "validation", "detail": f"images[{i}].data is not valid base64: {exc}"},
+                status_code=422,
+            )
+        decoded.append({"mime": img["mime"], "data": data})
+    return decoded, None
